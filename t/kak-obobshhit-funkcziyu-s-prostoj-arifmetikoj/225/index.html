#[derive(Debug, Clone, Copy)]
struct A<T = i32>(T);

#[derive(Debug, Clone, Copy)]
struct B<T = i32>(T);

fn a2b_f(a: A<f32>) -> B<f32> {
    B(-a.0 - a.0)
}

fn a2b_i(a: A) -> B {
    B(-a.0 - a.0)
}

// что должно быть на месте '?', что бы слепить две функции выше в одну обобщенную?
// fn a2b<T: ?>(a: A<T>) -> B<T> {
//     B(-a.0 - a.0)
// }

fn main() {
    let i_a = A(1);
    let i_b = a2b_i(i_a);
    println!("{:?} -> {:?}", i_a, i_b);
    
    let f_a = A(1.0);
    let f_b = a2b_f(f_a);
    println!("{:?} -> {:?}", f_a, f_b);
}

T: Copy + Debug + ops::Neg + ops::Sub не сильно помогли
