Сделал так:


            let global_timeout = poll_fn(move || -> Poll<Option<u8>, std::io::Error> {
                let timeout_poll = global_timeout.poll();
                match timeout_poll {
                    Ok(Async::Ready(())) => Err(std::io::Error::from(std::io::ErrorKind::TimedOut)),
                    Ok(Async::NotReady) => Ok(Async::NotReady),
                    Err(e) => Err(e),
                }
            });
