// compiles:
// fn power_level(pos @ (_, _): (u16, u16)) -> i64 { pos.0 as _ }

// fails to compile: "error[E0303]: pattern bindings are not allowed after an `@`":
fn power_level(pos @ (x, _): (u16, u16)) -> i64 { pos.0 as _ }

fn main() {
    let x = power_level((1, 2));
    println!("{}", x)
}
